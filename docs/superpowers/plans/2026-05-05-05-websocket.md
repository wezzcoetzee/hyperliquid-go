# Plan 05: WebSocket and Subscriptions Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement a single multiplexed WebSocket connection with auto-reconnect, ping/pong heartbeat, and typed subscription helpers for every Hyperliquid WS channel. Bonus: POST-over-WS for Info/Exchange.

**Architecture:** One `Conn` per `Client`, lazily opened on the first subscription. The connection has a writer-mutex, a single read goroutine, and a `subscriptionRegistry` mapping `(channel, args)` → list of subscriber callbacks. On reconnect, the registry is replayed (subscriptions auto-resume). Each subscription returns a `Subscription` handle with `Unsubscribe()`.

**Tech Stack:** Go 1.22+, `github.com/gorilla/websocket`, `golang.org/x/sync/errgroup`. Builds on Plan 01.

```go
import (
	"context"
	"sync"

	"github.com/gorilla/websocket"
)

type DefaultWSDialer struct{}

func (DefaultWSDialer) Dial(ctx context.Context, url string) (WSConn, error) {
	d := websocket.DefaultDialer
	c, _, err := d.DialContext(ctx, url, nil)
	if err != nil {
		return nil, err
	}
	return &defaultWSConn{c: c}, nil
}

type defaultWSConn struct {
	c     *websocket.Conn
	wmu   sync.Mutex
}

func (w *defaultWSConn) WriteJSON(v any) error {
	w.wmu.Lock()
	defer w.wmu.Unlock()
	return w.c.WriteJSON(v)
}

func (w *defaultWSConn) ReadMessage(ctx context.Context) ([]byte, error) {
	type res struct {
		b   []byte
		err error
	}
	ch := make(chan res, 1)
	go func() {
		_, b, err := w.c.ReadMessage()
		ch <- res{b, err}
	}()
	select {
	case r := <-ch:
		return r.b, r.err
	case <-ctx.Done():
		_ = w.c.Close()
		return nil, ctx.Err()
	}
}

func (w *defaultWSConn) Close() error { return w.c.Close() }
```

```go
package ws

import "sync"

type subscriber struct {
	id int64
	cb func([]byte)
}

type registry struct {
	mu    sync.RWMutex
	next  int64
	byKey map[string][]subscriber
	keys  map[string]any // remember subscription params for replay on reconnect
}

func newRegistry() *registry {
	return &registry{byKey: map[string][]subscriber{}, keys: map[string]any{}}
}

func (r *registry) add(key string, cb func([]byte)) int64 {
	r.mu.Lock()
	defer r.mu.Unlock()
	r.next++
	id := r.next
	r.byKey[key] = append(r.byKey[key], subscriber{id: id, cb: cb})
	return id
}

func (r *registry) remove(key string, id int64) (lastGone bool) {
	r.mu.Lock()
	defer r.mu.Unlock()
	subs := r.byKey[key]
	out := subs[:0]
	for _, s := range subs {
		if s.id != id {
			out = append(out, s)
		}
	}
	if len(out) == 0 {
		delete(r.byKey, key)
		delete(r.keys, key)
		return true
	}
	r.byKey[key] = out
	return false
}

func (r *registry) subscribers(key string) []subscriber {
	r.mu.RLock()
	defer r.mu.RUnlock()
	out := make([]subscriber, len(r.byKey[key]))
	copy(out, r.byKey[key])
	return out
}

func (r *registry) recordParams(key string, params any) {
	r.mu.Lock()
	defer r.mu.Unlock()
	r.keys[key] = params
}

func (r *registry) snapshot() map[string]any {
	r.mu.RLock()
	defer r.mu.RUnlock()
	out := make(map[string]any, len(r.keys))
	for k, v := range r.keys {
		out[k] = v
	}
	return out
}
```

```go
package ws

import (
	"context"
	"encoding/json"
	"errors"
	"sync"
	"time"

	"github.com/wezzcoetzee/hyperliquid-go/transport"
)

type Conn struct {
	url      string
	dialer   transport.WSDialer
	registry *registry

	mu      sync.Mutex
	conn    transport.WSConn
	closed  bool

	// signaling
	pingPeriod    time.Duration
	reconnectMin  time.Duration
	reconnectMax  time.Duration

	rootCtx context.Context
	cancel  context.CancelFunc
}

func newConn(parent context.Context, url string, d transport.WSDialer, r *registry) *Conn {
	ctx, cancel := context.WithCancel(parent)
	return &Conn{
		url: url, dialer: d, registry: r,
		pingPeriod: 30 * time.Second,
		reconnectMin: 500 * time.Millisecond, reconnectMax: 30 * time.Second,
		rootCtx: ctx, cancel: cancel,
	}
}

func (c *Conn) ensure(ctx context.Context) error {
	c.mu.Lock()
	defer c.mu.Unlock()
	if c.closed {
		return errors.New("ws: conn closed")
	}
	if c.conn != nil {
		return nil
	}
	conn, err := c.dialer.Dial(ctx, c.url)
	if err != nil {
		return err
	}
	c.conn = conn
	go c.readLoop()
	go c.pingLoop()
	go c.replaySubs()
	return nil
}

func (c *Conn) readLoop() {
	conn := c.conn
	for {
		msg, err := conn.ReadMessage(c.rootCtx)
		if err != nil {
			c.handleDisconnect()
			return
		}
		c.dispatch(msg)
	}
}

func (c *Conn) dispatch(msg []byte) {
	var env struct {
		Channel string          `json:"channel"`
		Data    json.RawMessage `json:"data"`
	}
	if err := json.Unmarshal(msg, &env); err != nil {
		return
	}
	// Channel -> key mapping handled in subs.go (e.g., "trades" + coin -> "trades:BTC").
	// We dispatch to all keys matching the channel; per-coin filtering happens in callbacks.
	for _, sub := range c.registry.subscribers(env.Channel) {
		sub.cb(env.Data)
	}
	// Also dispatch to keyed subscribers (e.g., "l2Book:BTC")
	for k, subs := range c.registry.allByKey() {
		if startsWith(k, env.Channel+":") {
			for _, s := range subs {
				s.cb(env.Data)
			}
		}
	}
}

func startsWith(s, prefix string) bool {
	return len(s) >= len(prefix) && s[:len(prefix)] == prefix
}

func (c *Conn) pingLoop() {
	t := time.NewTicker(c.pingPeriod)
	defer t.Stop()
	for {
		select {
		case <-c.rootCtx.Done():
			return
		case <-t.C:
			c.mu.Lock()
			conn := c.conn
			c.mu.Unlock()
			if conn == nil {
				return
			}
			_ = conn.WriteJSON(map[string]string{"method": "ping"})
		}
	}
}

func (c *Conn) handleDisconnect() {
	c.mu.Lock()
	if c.conn != nil {
		_ = c.conn.Close()
		c.conn = nil
	}
	closed := c.closed
	c.mu.Unlock()
	if closed {
		return
	}
	go c.reconnect()
}

func (c *Conn) reconnect() {
	backoff := c.reconnectMin
	for {
		select {
		case <-c.rootCtx.Done():
			return
		case <-time.After(backoff):
		}
		if err := c.ensure(c.rootCtx); err == nil {
			return
		}
		backoff *= 2
		if backoff > c.reconnectMax {
			backoff = c.reconnectMax
		}
	}
}

func (c *Conn) replaySubs() {
	for _, params := range c.registry.snapshot() {
		_ = c.WriteJSON(map[string]any{"method": "subscribe", "subscription": params})
	}
}

func (c *Conn) WriteJSON(v any) error {
	c.mu.Lock()
	conn := c.conn
	c.mu.Unlock()
	if conn == nil {
		return errors.New("ws: not connected")
	}
	return conn.WriteJSON(v)
}

func (c *Conn) Close() {
	c.mu.Lock()
	c.closed = true
	if c.conn != nil {
		_ = c.conn.Close()
	}
	c.mu.Unlock()
	c.cancel()
}
```

```go
package ws

import (
	"context"
	"sync"

	"github.com/wezzcoetzee/hyperliquid-go/transport"
)

type Client struct {
	URL    string
	Dialer transport.WSDialer

	once     sync.Once
	conn     *Conn
	registry *registry
}

func (c *Client) ensureStarted(ctx context.Context) error {
	c.once.Do(func() {
		if c.Dialer == nil {
			c.Dialer = transport.DefaultWSDialer{}
		}
		c.registry = newRegistry()
		c.conn = newConn(ctx, c.URL, c.Dialer, c.registry)
	})
	return c.conn.ensure(ctx)
}

type Subscription struct {
	c   *Client
	key string
	id  int64
	sub map[string]any
}

func (s *Subscription) Unsubscribe(ctx context.Context) error {
	last := s.c.registry.remove(s.key, s.id)
	if last {
		return s.c.conn.WriteJSON(map[string]any{"method": "unsubscribe", "subscription": s.sub})
	}
	return nil
}

func (c *Client) subscribe(ctx context.Context, key string, params map[string]any, cb func([]byte)) (*Subscription, error) {
	if err := c.ensureStarted(ctx); err != nil {
		return nil, err
	}
	id := c.registry.add(key, cb)
	c.registry.recordParams(key, params)
	if err := c.conn.WriteJSON(map[string]any{"method": "subscribe", "subscription": params}); err != nil {
		c.registry.remove(key, id)
		return nil, err
	}
	return &Subscription{c: c, key: key, id: id, sub: params}, nil
}
```

- Each POST is wrapped: `{method: "post", id, request: {type: "info"|"action", payload}}`
- A response router maps `id` → channel waiting for the reply.

Implement:

```go
type postRouter struct {
	mu      sync.Mutex
	next    uint64
	waiters map[uint64]chan []byte
}

func (p *postRouter) call(ctx context.Context, c *Conn, kind string, payload any) ([]byte, error) {
	p.mu.Lock()
	p.next++
	id := p.next
	ch := make(chan []byte, 1)
	if p.waiters == nil {
		p.waiters = map[uint64]chan []byte{}
	}
	p.waiters[id] = ch
	p.mu.Unlock()

	if err := c.WriteJSON(map[string]any{"method": "post", "id": id, "request": map[string]any{"type": kind, "payload": payload}}); err != nil {
		return nil, err
	}
	select {
	case b := <-ch:
		return b, nil
	case <-ctx.Done():
		return nil, ctx.Err()
	}
}
```
