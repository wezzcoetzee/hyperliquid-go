# Plan 01: Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Scaffold the `github.com/wezzcoetzee/hyperliquid` module with a constructible `Client`, network config, error types, shared domain types, and pluggable HTTP transport. End state: an importable library that can POST to `/info` via a stub method and round-trip JSON.

**Architecture:** A top-level `Client` struct holds three method-group fields (`Info`, `Exchange`, `Subscriptions`) wired to a shared `transport.HTTP` and (optionally) a `Signer`. Method groups are tiny structs sharing a pointer back to the client. Network endpoints are constants resolved through a `Network` enum. All HTTP I/O goes through a single `transport.HTTP` interface with a default `net/http`-backed implementation.

**Tech Stack:** Go 1.22+, stdlib only for this plan (no third-party deps yet).

```go
type ActionError struct {
	Action   string
	Response string
}

	return fmt.Sprintf("hyperliquid: action %q rejected: %s", e.Action, e.Response)
}

type SignError struct {
	Err error
}

func (e *SignError) Error() string { return "hyperliquid: sign error: " + e.Err.Error() }
func (e *SignError) Unwrap() error { return e.Err }

type WSError struct {
	Code   int
	Reason string
}

func (e *WSError) Error() string {
	return fmt.Sprintf("hyperliquid: websocket error %d: %s", e.Code, e.Reason)
}
```

```go
package types

import (
	"encoding/hex"
	"errors"
	"strings"
)

type Address [20]byte

func ParseAddress(s string) (Address, error) {
	var a Address
	s = strings.TrimPrefix(strings.ToLower(s), "0x")
	if len(s) != 40 {
		return a, errors.New("invalid address length")
	}
	b, err := hex.DecodeString(s)
	if err != nil {
		return a, err
	}
	copy(a[:], b)
	return a, nil
}

func (a Address) Hex() string {
	return "0x" + hex.EncodeToString(a[:])
}

func (a Address) String() string { return a.Hex() }

type Side string

const (
	Buy  Side = "B"
	Sell Side = "A"
)

func (s Side) String() string { return string(s) }

type Tif string

const (
	TifGtc Tif = "Gtc"
	TifIoc Tif = "Ioc"
	TifAlo Tif = "Alo"
)
```

`transport/http.go`:

```go
package transport

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"time"
)

type HTTP interface {
	PostJSON(ctx context.Context, path string, body any, out any) error
}

type DefaultHTTP struct {
	baseURL string
	client  *http.Client
}

func NewDefaultHTTP(baseURL string, client *http.Client) *DefaultHTTP {
	if client == nil {
		client = &http.Client{Timeout: 30 * time.Second}
	}
	return &DefaultHTTP{baseURL: baseURL, client: client}
}

func (h *DefaultHTTP) PostJSON(ctx context.Context, path string, body any, out any) error {
	buf, err := json.Marshal(body)
	if err != nil {
		return err
	}
	req, err := http.NewRequestWithContext(ctx, "POST", h.baseURL+path, bytes.NewReader(buf))
	if err != nil {
		return err
	}
	req.Header.Set("Content-Type", "application/json")
	resp, err := h.client.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	respBody, err := io.ReadAll(resp.Body)
	if err != nil {
		return err
	}
	if resp.StatusCode/100 != 2 {
		return &TransportAPIError{Status: resp.StatusCode, Body: string(respBody)}
	}
	if out == nil {
		return nil
	}
	if err := json.Unmarshal(respBody, out); err != nil {
		return fmt.Errorf("decode: %w (body=%s)", err, string(respBody))
	}
	return nil
}

type TransportAPIError struct {
	Status int
	Body   string
}

func (e *TransportAPIError) Error() string {
	return fmt.Sprintf("http %d: %s", e.Status, e.Body)
}
```

> Note: `transport` package returns its own low-level error to avoid an import cycle with the root package. The root package's `Client` wraps this into `*hyperliquid.APIError` at the boundary.

```go
package hyperliquid

import (
	"net/http"

	"github.com/wezzcoetzee/hyperliquid/exchange"
	"github.com/wezzcoetzee/hyperliquid/info"
	"github.com/wezzcoetzee/hyperliquid/transport"
	"github.com/wezzcoetzee/hyperliquid/ws"
)

type Signer interface {
	// Filled in plan 02. Declared here so Config can reference it without an import cycle.
}

type Config struct {
	Network Network
	Signer  Signer
	HTTP    *http.Client
	BaseURL string // overrides Network.HTTPURL() — used for tests/proxies
	WSURL   string // overrides Network.WSURL()
}

type Client struct {
	Network       Network
	Info          *info.Client
	Exchange      *exchange.Client
	Subscriptions *ws.Client
}

func New(cfg Config) (*Client, error) {
	baseURL := cfg.BaseURL
	if baseURL == "" {
		baseURL = cfg.Network.HTTPURL()
	}
	httpTr := transport.NewDefaultHTTP(baseURL, cfg.HTTP)
	return &Client{
		Network:       cfg.Network,
		Info:          &info.Client{HTTP: httpTr},
		Exchange:      &exchange.Client{HTTP: httpTr},
		Subscriptions: &ws.Client{},
	}, nil
}
```
