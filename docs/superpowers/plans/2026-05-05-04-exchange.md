# Plan 04: Exchange Client Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement every signed `/exchange` action with typed parameters, signing pipeline integration, response decoding, and per-action signing fixtures captured from the TS SDK.

**Architecture:** Each action is a method on `exchange.Client`. The flow is identical for all actions:
1. Build the action as a `*msgpack.OrderedMap` with TS-equivalent key ordering.
2. Generate nonce (millisecond timestamp, monotonic with mutex).
3. For L1 actions: `BuildL1Signature(...)`. For user-signed: `BuildUserSignature(...)`.
4. POST `{action, nonce, signature, vaultAddress?}` to `/exchange`.
5. Decode the typed response and convert `{status: "err"}` to `*ActionError`.

A shared `submit` helper handles steps 2–5. Each method only constructs the action.

**Tech Stack:** Go 1.22+. Builds on Plans 01–02.

```go
package exchange

import (
	"sync"
	"time"
)

type nonceGen struct {
	mu   sync.Mutex
	last uint64
}

func (g *nonceGen) next() uint64 {
	g.mu.Lock()
	defer g.mu.Unlock()
	now := uint64(time.Now().UnixMilli())
	if now <= g.last {
		now = g.last + 1
	}
	g.last = now
	return now
}
```

```go
package exchange

import (
	"context"
	"encoding/json"
	"errors"

	hl "github.com/wezzcoetzee/hyperliquid-go"
	"github.com/wezzcoetzee/hyperliquid-go/internal/msgpack"
	"github.com/wezzcoetzee/hyperliquid-go/signer"
	"github.com/wezzcoetzee/hyperliquid-go/transport"
)

type Client struct {
	HTTP    transport.HTTP
	Signer  signer.Signer
	Network hl.Network

	// VaultAddress, if set, signs every action on behalf of the vault.
	VaultAddress *[20]byte

	nonces nonceGen
}

// submitL1 signs and submits an L1 (Exchange) action.
func (c *Client) submitL1(ctx context.Context, action *msgpack.OrderedMap, out any) error {
	if c.Signer == nil {
		return errors.New("hyperliquid/exchange: Signer required for write operations")
	}
	nonce := c.nonces.next()
	var vault []byte
	if c.VaultAddress != nil {
		vault = c.VaultAddress[:]
	}
	sig, err := BuildL1Signature(ctx, c.Signer, action, nonce, vault, c.Network == hl.Mainnet)
	if err != nil {
		return err
	}
	return c.send(ctx, action, nonce, sig, vault, out)
}

// submitUser signs and submits a user-signed action.
func (c *Client) submitUser(ctx context.Context, action *msgpack.OrderedMap, primaryType string, fields []signer.Field, message map[string]any, out any) error {
	if c.Signer == nil {
		return errors.New("hyperliquid/exchange: Signer required for write operations")
	}
	nonce, _ := message["time"].(uint64) // user-signed actions embed time in payload
	if nonce == 0 {
		nonce = c.nonces.next()
		message["time"] = nonce
	}
	chainID := uint64(42161)
	if c.Network == hl.Testnet {
		chainID = 421614
	}
	sig, err := BuildUserSignature(ctx, c.Signer, primaryType, fields, message, chainID)
	if err != nil {
		return err
	}
	return c.send(ctx, action, nonce, sig, nil, out)
}

func (c *Client) send(ctx context.Context, action *msgpack.OrderedMap, nonce uint64, sig signer.Signature, vault []byte, out any) error {
	body := map[string]any{
		"action":    actionMapAsJSON(action),
		"nonce":     nonce,
		"signature": map[string]any{"r": "0x" + hexEnc(sig.R[:]), "s": "0x" + hexEnc(sig.S[:]), "v": int(sig.V)},
	}
	if len(vault) > 0 {
		body["vaultAddress"] = "0x" + hexEnc(vault)
	}
	var raw StatusResponse
	if err := c.HTTP.PostJSON(ctx, "/exchange", body, &raw); err != nil {
		return err
	}
	if raw.Status != "ok" {
		return &hl.ActionError{Action: actionType(action), Response: string(raw.Response)}
	}
	if out != nil {
		return json.Unmarshal(raw.Response, out)
	}
	return nil
}

func actionType(m *msgpack.OrderedMap) string {
	// OrderedMap doesn't expose Get; add accessor to msgpack package OR walk Keys().
	// (Add `func (m *OrderedMap) Get(k string) (any, bool)` to msgpack package as part of this task.)
	v, _ := m.Get("type")
	s, _ := v.(string)
	return s
}

// actionMapAsJSON converts an OrderedMap to a JSON-encodable structure preserving insertion order.
// json.Marshal(map[string]any) randomizes; we render manually.
func actionMapAsJSON(m *msgpack.OrderedMap) json.RawMessage {
	b, _ := orderedToJSON(m)
	return b
}
```

Create `internal/msgpack/json.go`:

```go
package msgpack

import (
	"bytes"
	"encoding/json"
)

// MarshalOrderedJSON renders an OrderedMap (and nested values) as JSON, preserving key order.
func MarshalOrderedJSON(v any) ([]byte, error) {
	var buf bytes.Buffer
	if err := writeJSON(&buf, v); err != nil {
		return nil, err
	}
	return buf.Bytes(), nil
}

func writeJSON(buf *bytes.Buffer, v any) error {
	switch x := v.(type) {
	case *OrderedMap:
		buf.WriteByte('{')
		for i, k := range x.keys {
			if i > 0 {
				buf.WriteByte(',')
			}
			kb, _ := json.Marshal(k)
			buf.Write(kb)
			buf.WriteByte(':')
			if err := writeJSON(buf, x.values[k]); err != nil {
				return err
			}
		}
		buf.WriteByte('}')
		return nil
	case []any:
		buf.WriteByte('[')
		for i, e := range x {
			if i > 0 {
				buf.WriteByte(',')
			}
			if err := writeJSON(buf, e); err != nil {
				return err
			}
		}
		buf.WriteByte(']')
		return nil
	default:
		b, err := json.Marshal(v)
		if err != nil {
			return err
		}
		buf.Write(b)
		return nil
	}
}
```

```go
package exchange

import (
	"context"

	"github.com/wezzcoetzee/hyperliquid-go/internal/msgpack"
)

type Tif string

const (
	TifGtc Tif = "Gtc"
	TifIoc Tif = "Ioc"
	TifAlo Tif = "Alo"
)

type LimitOrder struct {
	Tif Tif
}

type TriggerOrder struct {
	IsMarket bool
	TriggerPx string
	Tpsl     string // "tp" | "sl"
}

type OrderType struct {
	Limit   *LimitOrder
	Trigger *TriggerOrder
}

type OrderParams struct {
	Asset      uint32 // perp asset index, or 10000+spotIndex for spot
	IsBuy      bool
	LimitPx    string
	Sz         string
	ReduceOnly bool
	OrderType  OrderType
	Cloid      string // optional 16-byte hex prefixed with 0x
}

type OrderRequest struct {
	Orders   []OrderParams
	Grouping string // "na" | "normalTpsl" | "positionTpsl"
	Builder  *Builder
}

type Builder struct {
	Address string // 0x...
	Fee     int    // tenths of a basis point
}

func buildOrderAction(req OrderRequest) *msgpack.OrderedMap {
	a := msgpack.NewOrderedMap()
	a.Set("type", "order")

	orders := make([]any, len(req.Orders))
	for i, o := range req.Orders {
		om := msgpack.NewOrderedMap()
		om.Set("a", uint64(o.Asset))
		om.Set("b", o.IsBuy)
		om.Set("p", o.LimitPx)
		om.Set("s", o.Sz)
		om.Set("r", o.ReduceOnly)

		t := msgpack.NewOrderedMap()
		switch {
		case o.OrderType.Limit != nil:
			lim := msgpack.NewOrderedMap()
			lim.Set("tif", string(o.OrderType.Limit.Tif))
			t.Set("limit", lim)
		case o.OrderType.Trigger != nil:
			trg := msgpack.NewOrderedMap()
			trg.Set("isMarket", o.OrderType.Trigger.IsMarket)
			trg.Set("triggerPx", o.OrderType.Trigger.TriggerPx)
			trg.Set("tpsl", o.OrderType.Trigger.Tpsl)
			t.Set("trigger", trg)
		}
		om.Set("t", t)
		if o.Cloid != "" {
			om.Set("c", o.Cloid)
		}
		orders[i] = om
	}
	a.Set("orders", orders)
	a.Set("grouping", req.Grouping)
	if req.Builder != nil {
		b := msgpack.NewOrderedMap()
		b.Set("b", req.Builder.Address)
		b.Set("f", uint64(req.Builder.Fee))
		a.Set("builder", b)
	}
	return a
}

type OrderResponse struct {
	Statuses []OrderStatus
}

func (c *Client) Order(ctx context.Context, req OrderRequest) (*OrderResponse, error) {
	action := buildOrderAction(req)
	var inner OrderResponseInner
	if err := c.submitL1(ctx, action, &inner); err != nil {
		return nil, err
	}
	return &OrderResponse{Statuses: inner.Data.Statuses}, nil
}
```
