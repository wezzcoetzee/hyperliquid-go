# Plan 03: Info Client Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement every read-only `/info` endpoint with typed request/response, table-driven tests against an `httptest.Server`, and a one-time integration smoke test against testnet.

**Architecture:** Each endpoint is a single method on `info.Client` that posts a typed payload to `/info` and decodes a typed response. We establish the pattern with one method (TDD), then extend it across all endpoints in batched tasks. Per-method tests use `httptest.Server` with canned JSON responses captured from the public Hyperliquid API.

**Tech Stack:** Go 1.22+, stdlib only. Builds on Plans 01 and 02.

Split by domain because Hyperliquid groups endpoints semantically; each file stays under ~400 lines.

```go
package info

import (
	"context"
	"encoding/json"
	"io"
	"net/http"
	"net/http/httptest"
	"os"
	"path/filepath"
	"testing"

	"github.com/wezzcoetzee/hyperliquid-go/transport"
)

func newTestClient(t *testing.T, handler http.HandlerFunc) *Client {
	t.Helper()
	srv := httptest.NewServer(handler)
	t.Cleanup(srv.Close)
	return &Client{HTTP: transport.NewDefaultHTTP(srv.URL, nil)}
}

func loadResponse(t *testing.T, name string) []byte {
	t.Helper()
	b, err := os.ReadFile(filepath.Join("testdata", "responses", name))
	if err != nil {
		t.Fatal(err)
	}
	return b
}

func TestClient_AllMids(t *testing.T) {
	c := newTestClient(t, func(w http.ResponseWriter, r *http.Request) {
		var req map[string]any
		body, _ := io.ReadAll(r.Body)
		_ = json.Unmarshal(body, &req)
		if req["type"] != "allMids" {
			t.Fatalf("unexpected type: %v", req["type"])
		}
		w.Header().Set("Content-Type", "application/json")
		_, _ = w.Write(loadResponse(t, "all_mids.json"))
	})

	mids, err := c.AllMids(context.Background())
	if err != nil {
		t.Fatal(err)
	}
	if mids["BTC"] != "30000.5" {
		t.Fatalf("BTC = %v", mids["BTC"])
	}
}
```

func (c *Client) MetaAndAssetCtxs(ctx context.Context) (*Meta, []AssetCtx, error) {
	var out []json.RawMessage
	if err := c.post(ctx, map[string]any{"type": "metaAndAssetCtxs"}, &out); err != nil {
		return nil, nil, err
	}
	if len(out) != 2 {
		return nil, nil, fmt.Errorf("expected 2 elements, got %d", len(out))
	}
	var meta Meta
	var ctxs []AssetCtx
	if err := json.Unmarshal(out[0], &meta); err != nil {
		return nil, nil, err
	}
	if err := json.Unmarshal(out[1], &ctxs); err != nil {
		return nil, nil, err
	}
	return &meta, ctxs, nil
}
